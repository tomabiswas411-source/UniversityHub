   462→async function seedSchedules(
   463→  ctx: Awaited<ReturnType<typeof seedUniversityAndDepartments>>,
   464→  users: Awaited<ReturnType<typeof seedUsers>>,
   465→  academics: Awaited<ReturnType<typeof seedAcademics>>
   466→) {
   467→  const { ice, cse, math, iceBsc, iceMsc, cseBsc, mathBsc } = ctx;
   468→  const { iceTeachers, cseTeachers, mathTeachers } = users;
   469→  const { iceCourses, cseCourses, mathCourses, iceRooms, cseRooms, mathRooms, iceSlots, cseSlots, mathSlots } = academics;
   470→
   471→  let total = 0;
   472→  const busyTeacher = new Set<string>(); // `${teacherId}|${day}|${slotIdx}`
   473→  const busyRoom = new Set<string>(); // `${roomId}|${day}|${slotIdx}`
   474→
   475→  async function place(
   476→    departmentId: string,
   477→    programId: string,
   478→    semester: number,
   479→    course: { id: string; code: string; name: string; type: string },
   480→    teachers: { id: string; fullName: string }[],
   481→    rooms: { id: string; roomNumber: string }[],
   482→    slots: { id: string; startTime: string; endTime: string; isBreak: boolean }[],
   483→    day: string,
   484→    slotIdx: number,
   485→    createdById: string
   486→  ): Promise<boolean> {
   487→    const slot = slots[slotIdx];
   488→    if (!slot || slot.isBreak) return false;
   489→
   490→    // find free teacher (prefer one whose subject area fits by round robin)
   491→    let chosenTeacher = null;
   492→    for (const t of teachers) {
   493→      const key = `${t.id}|${day}|${slotIdx}`;
   494→      if (!busyTeacher.has(key)) {
   495→        chosenTeacher = t;
   496→        break;
   497→      }
   498→    }
   499→    if (!chosenTeacher) return false;
   500→
   501→    // find free room — labs need lab rooms
   502→    let chosenRoom = null;
   503→    for (const r of rooms) {
   504→      const key = `${r.id}|${day}|${slotIdx}`;
   505→      const isLabRoom = r.roomNumber.toLowerCase().includes("lab");
   506→      if (course.type === "lab" && !isLabRoom) continue;
   507→      if (course.type !== "lab" && isLabRoom) continue;
   508→      if (!busyRoom.has(key)) {
   509→        chosenRoom = r;
   510→        break;
   511→      }
   512→    }
   513→    if (!chosenRoom) return false;
   514→
   515→    busyTeacher.add(`${chosenTeacher.id}|${day}|${slotIdx}`);
   516→    busyRoom.add(`${chosenRoom.id}|${day}|${slotIdx}`);
   517→
   518→    await db.schedule.create({
   519→      data: {
   520→        departmentId,
   521→        programId,
   522→        semester,
   523→        courseId: course.id,
   524→        teacherId: chosenTeacher.id,
   525→        roomId: chosenRoom.id,
   526→        timeSlotId: slot.id,
