    89→/* ---------------- local helpers ---------------- */

    91→function currentDay(): DayOfWeek {
    92→  return new Date()
    93→    .toLocaleDateString("en-US", { weekday: "long" })
    94→    .toLowerCase() as DayOfWeek;
    95→}
    96→
    97→function fmtDate(iso: string): string {
    98→  return new Date(iso).toLocaleDateString("en-US", { month: "short", day: "numeric" });
    99→}
   100→
   101→function fmtTime(t: string): string {
   102→  return t.length >= 5 ? t.slice(0, 5) : t;
   103→}
   104→
   105→function initials(name: string): string {
   106→  const parts = name.trim().split(/\s+/).slice(0, 2);
   107→  return (parts.map((w) => w[0]?.toUpperCase() ?? "").join("") || "S");
   108→}

   125→function gradeBadgeCls(grade: string | null): string {
   126→  if (!grade) return "bg-muted text-muted-foreground";
   127→  if (grade.startsWith("A+"))
   128→    return "bg-emerald-100 text-emerald-700 dark:bg-emerald-950/50 dark:text-emerald-300";
   129→  if (grade.startsWith("A"))
   130→    return "bg-violet-100 text-violet-700 dark:bg-violet-950/50 dark:text-violet-300";
   131→  if (grade.startsWith("B"))
   132→    return "bg-amber-100 text-amber-700 dark:bg-amber-950/50 dark:text-amber-300";
   133→  if (grade.startsWith("F"))
   134→    return "bg-rose-100 text-rose-700 dark:bg-rose-950/50 dark:text-rose-300";
   135→  return "bg-muted text-muted-foreground"; // C / D
   136→}
   137→
   138→function dueChip(dueISO: string | null): { label: string; cls: string } {
   139→  if (!dueISO) return { label: "No due date", cls: "bg-muted text-muted-foreground" };
   140→  const due = new Date(dueISO);
   141→  const now = new Date();
   142→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   143→  const startOfDue = new Date(due.getFullYear(), due.getMonth(), due.getDate());
   144→  const days = Math.round((startOfDue.getTime() - startOfToday.getTime()) / 86_400_000);
   145→  if (days < 0)
   146→    return { label: "Overdue", cls: "bg-rose-100 text-rose-700 dark:bg-rose-950/50 dark:text-rose-300" };
   147→  if (days <= 3)
   148→    return {
   149→      label: days === 0 ? "Due today" : `Due in ${days}d`,
   150→      cls: "bg-amber-100 text-amber-700 dark:bg-amber-950/50 dark:text-amber-300",
   151→    };
   152→  return { label: fmtDate(dueISO), cls: "bg-muted text-muted-foreground" };
   153→}

   166→function ChangeRow({ change }: { change: ScheduleChangeDTO }) {
   167→  const details: string[] = [];
   168→  if (change.newDay) details.push(DAY_LABELS[change.newDay as DayOfWeek] ?? change.newDay);
   169→  if (change.newStartTime)
   170→    details.push(
   171→      `${fmtTime(change.newStartTime)}${change.newEndTime ? `–${fmtTime(change.newEndTime)}` : ""}`
   172→    );
   173→  if (change.newRoomNumber) details.push(`Room ${change.newRoomNumber}`);
   174→
   175→  return (
   176→    <div
   177→      className={cn(
   178→        "rounded-xl border bg-background/60 p-3 border-l-4 sm:p-4",
   179→        CHANGE_BORDER[change.changeType] ?? "border-l-foreground/25"
   180→      )}
   181→    >
   182→      <div className="flex flex-wrap items-center gap-2">
   183→        <ChangeTypeBadge type={change.changeType} />
   184→        <span className="text-sm font-bold">{change.courseCode}</span>
   185→        {change.courseName && (
   186→          <span className="min-w-0 truncate text-xs text-muted-foreground">{change.courseName}</span>
   187→        )}
   188→      </div>
   189→      <p className="mt-1 text-sm text-muted-foreground">{change.reason}</p>
   190→      <div className="mt-2 flex flex-wrap items-center gap-x-3 gap-y-1 text-[11px] text-muted-foreground">
   191→        <span className="inline-flex items-center gap-1">
   192→          <CalendarDays className="h-3 w-3" aria-hidden />
   193→          Effective {fmtDate(change.effectiveDate)}
   194→        </span>
   195→        {details.length > 0 && (
   196→          <span className="inline-flex items-center gap-1">
   197→            <MapPin className="h-3 w-3" aria-hidden />
   198→            {details.join(" · ")}
   199→          </span>
   200→        )}
   201→      </div>
   202→    </div>
   203→  );
   204→}

   285→    return [...(data?.changes ?? [])].sort(
   286→      (a, b) =>
